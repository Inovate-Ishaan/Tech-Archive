  const [alert, setAlert] = useState(null);
  const alertRef =  useRef(null);

  function showAlert(type, msg) {
    setAlert({
      type,
      msg,
    });

    clearTimeout(alertRef.current);

    alertRef.current = setTimeout(() => {
      setAlert(null);
    }, 3000);
  }

  const closeAlert = () => {
    setAlert(null);
  };
